⚡ AUTOMATIC INSTALLATION (1 COMMAND)
──────────────────────────────────────────────────────────────
cd /path/to/Binance-2
chmod +x install_ubuntu.sh
./install_ubuntu.sh


📋 MANUAL INSTALLATION (5 STEPS)
──────────────────────────────────────────────────────────────
# 1. Update system
sudo apt-get update && sudo apt-get install -y python3 python3-venv python3-pip

# 2. Create virtual environment
python3 -m venv venv
source venv/bin/activate

# 3. Install packages
pip install -r requirements.txt

# 4. Configure
cp .env.example .env
nano .env  # Add Telegram credentials

# 5. Test
python test_setup.py


🚀 RUNNING THE BOT
──────────────────────────────────────────────────────────────
# Option 1: Direct (testing)
source venv/bin/activate
python main.py --once

# Option 2: Background with Screen
screen -S crypto-bot
source venv/bin/activate
python main.py
# Detach: Ctrl+A, D
# Reattach: screen -r crypto-bot

# Option 3: Systemd Service (production)
./install_ubuntu.sh  # Will setup service
sudo systemctl start binance-bot
sudo systemctl enable binance-bot


📊 MONITORING
──────────────────────────────────────────────────────────────
# Check status
sudo systemctl status binance-bot

# View logs (live)
sudo journalctl -u binance-bot -f

# View log file
tail -f bot.log

──────────────────────────────────────────────────────────────
